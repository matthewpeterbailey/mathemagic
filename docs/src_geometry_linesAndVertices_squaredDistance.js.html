 * Returns the squared euclidean distance between points a and b
 * @param a Point A
 * @param b Point B
 */
var squaredDistance = function (a, b) {
    var deltaX = b.x - a.x;
    var deltaY = b.y - a.y;
    return deltaX * deltaX + deltaY * deltaY;
};
